// 移除URL中的多余参数
function cleanUrl(url) {
if (!url) return '';
// 移除?及后面的所有内容
return url.split('?')[0];
}
// 历史记录功能
var historyIPs = JSON.parse(localStorage.getItem('historyIPs')) || [];
var historyList = document.getElementById('historyList');
var resetButton = document.getElementById('resetButton');
populateHistoryList(historyIPs);
function populateHistoryList(ips) {
historyList.innerHTML = '';
ips.forEach(function(ip) {
var option = document.createElement('option');
option.value = ip;
historyList.appendChild(option);
});
}
function storeIPToHistory(ip) {
if (ip && ip.trim() !== '' && /^(?:[0-9]{1,3}\.){3}[0-9]{1,3}(?::\d{1,5})?$/.test(ip)) {
if (historyIPs.length >= 5) {
historyIPs.pop();
}
if (!historyIPs.includes(ip)) {
historyIPs.unshift(ip);
localStorage.setItem('historyIPs', JSON.stringify(historyIPs));
populateHistoryList(historyIPs);
}
}
}
// 表单提交处理
document.getElementById('myForm').addEventListener('submit', function(e) {
e.preventDefault();
var src = document.getElementById('inputSrc').value;
var resultContainer = document.getElementById('result');
// 显示加载指示器
document.getElementById('loadingIndicator').classList.remove('hidden');
// 使用代码2的API调用方式
var xhr = new XMLHttpRequest();
xhr.open('POST', '', true);
xhr.setRequestHeader('Content-Type', 'application/x-www-form-urlencoded');
xhr.onload = function() {
// 隐藏加载指示器
document.getElementById('loadingIndicator').classList.add('hidden');
if (xhr.status === 200) {
// 处理响应
resultContainer.innerHTML = xhr.responseText;
// 如果有结果，存储IP到历史记录
if (resultContainer.textContent.trim() !== '未找到酒店直播源') {
storeIPToHistory(src);
}
} else {
resultContainer.innerHTML = '<div class="text-red-500 p-4"><i class="fa fa-exclamation-circle mr-2"></i>请求出错，状态码：' + xhr.status + '</div>';
}
};
xhr.send('src=' + encodeURIComponent(src));
});
resetButton.addEventListener('click', function() {
document.getElementById('inputSrc').value = '';
});
// 一键下载按钮功能
document.getElementById('downloadBtn').addEventListener('click', function() {
const resultContainer = document.getElementById('result');
const text = resultContainer.textContent || resultContainer.innerText;
if (text.trim()) {
const fileName = 'hotel_sources.txt';
const blob = new Blob([text], { type: 'text/plain;charset=utf-8' });
const link = document.createElement('a');
link.href = URL.createObjectURL(blob);
link.download = fileName;
link.style.display = 'none';
document.body.appendChild(link);
link.click();
document.body.removeChild(link);
// 显示下载成功的提示
const originalText = this.innerHTML;
this.innerHTML = '<i class="fa fa-check mr-1"></i> 下载成功';
setTimeout(() => {
this.innerHTML = originalText;
}, 2000);
} else {
alert('没有可下载的内容，请先获取直播源');
}
});
// 添加刷新按钮功能
document.getElementById('refreshBtn').addEventListener('click', function() {
const currentSrc = document.getElementById('inputSrc').value;
if (currentSrc) {
// 模拟表单提交
const event = new Event('submit', { cancelable: true });
document.getElementById('myForm').dispatchEvent(event);
} else {
alert('请先输入IP地址和端口');
}
});
// 为结果容器添加平滑滚动
document.getElementById('resultContainer').style.scrollBehavior = 'smooth';
